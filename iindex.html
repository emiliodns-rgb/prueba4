        <div class="card">

        <div class="card">

            <h3>Registrar Nueva Venta</h3>

            

            <div class="input-group">

                <label>Cliente / Deudor</label>

    <script>

        // --- ESTADO DE LA APLICACIÓN ---

        let db = {

            productos: JSON.parse(localStorage.getItem('db_productos')) || [],

            clientes: JSON.parse(localStorage.getItem('db_clientes')) || []

        };


        let clienteSeleccionadoId = null;


        function guardarStorage() {

            localStorage.setItem('db_productos', JSON.stringify(db.productos));

            localStorage.setItem('db_clientes', JSON.stringify(db.clientes));

            render();

        }


        // --- NAVEGACIÓN ---

        function switchTab(tab) {

            document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));

            document.querySelectorAll('.nav-tabs button').forEach(b => b.classList.remove('active'));

            

            const tabBtn = document.getElementById(`tab-${tab}`);

            if (tabBtn) tabBtn.classList.add('active');

            

            document.getElementById(`view-${tab}`).classList.add('active');

            render();

        }


        // --- PRODUCTOS ---

        function guardarProducto() {

            const nombre = document.getElementById('prod-nombre').value.trim();

            const precio = parseFloat(document.getElementById('prod-precio').value);


            if (!nombre || isNaN(precio) || precio <= 0) {

                alert('Ingrese un nombre y un precio válido');

                return;

            }


            db.productos.push({ id: Date.now(), nombre, precio });

            document.getElementById('prod-nombre').value = '';

            document.getElementById('prod-precio').value = '';

            guardarStorage();

        }


        // --- VENTAS / COMPRAS ---

        function abrirModalNuevaVenta() {

            if (db.productos.length === 0) {

                alert('Primero debe agregar productos en la pestaña "Productos".');

                switchTab('productos');

                return;

            }


            const select = document.getElementById('venta-producto');

            select.innerHTML = db.productos.map(p => `<option value="${p.id}">${p.nombre} - $${p.precio}</option>`).join('');

            

            document.getElementById('venta-cliente').value = '';

            document.getElementById('venta-cantidad').value = 1;

            actualizarSubtotalVenta();

            

            document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));

            document.getElementById('modal-venta').classList.add('active');

        }


        function actualizarSubtotalVenta() {

            const prodId = parseInt(document.getElementById('venta-producto').value);

            const cant = parseInt(document.getElementById('venta-cantidad').value) || 0;

            const prod = db.productos.find(p => p.id === prodId);

            

            const total = prod ? prod.precio * cant : 0;

            document.getElementById('venta-total').innerText = `$${total.toLocaleString()}`;

        }


        function confirmarVenta() {

            const nombreCliente = document.getElementById('venta-cliente').value.trim();

            const prodId = parseInt(document.getElementById('venta-producto').value);

            const cantidad = parseInt(document.getElementById('venta-cantidad').value);


            if (!nombreCliente || isNaN(cantidad) || cantidad <= 0) {

                alert('Ingrese el nombre del cliente y una cantidad válida');

                return;

            }


            const prod = db.productos.find(p => p.id === prodId);

            const montoTotal = prod.precio * cantidad;


            // Buscar si el cliente ya existe activo

            let cliente = db.clientes.find(c => c.nombre.toLowerCase() === nombreCliente.toLowerCase() && c.activo);


            if (!cliente) {

                cliente = {

                    id: Date.now(),

                    nombre: nombreCliente,

                    activo: true,

                    movimientos: []

                };

                db.clientes.push(cliente);

            }


            // Registrar movimiento con fecha automática

            cliente.movimientos.push({

                tipo: 'COMPRA',

                detalle: `${prod.nombre} (x${cantidad})`,

                monto: montoTotal,

                fecha: new Date().toLocaleString()

            });


            guardarStorage();

            switchTab('inicio');

        }


        // --- PAGOS Y DETALLE ---

        function verDetalleCliente(id) {

            clienteSeleccionadoId = id;

            const cliente = db.clientes.find(c => c.id === id);

            

            document.getElementById('det-nombre').innerText = cliente.nombre;

            

            document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));

            document.getElementById('view-detalle').classList.add('active');

            

            renderDetalle();

        }


        function calcularDeudaCliente(cliente) {

            return cliente.movimientos.reduce((acc, m) => {

                return m.tipo === 'COMPRA' ? acc + m.monto : acc - m.monto;

            }, 0);

        }


        function ingresarAbono() {

            const montoInput = document.getElementById('monto-abono');

            const monto = parseFloat(montoInput.value);

            const cliente = db.clientes.find(c => c.id === clienteSeleccionadoId);


            if (isNaN(monto) || monto <= 0) {

                alert('Ingrese un monto válido');

                return;

            }


            cliente.movimientos.push({

                tipo: 'PAGO',

                detalle: 'Abono realizado',

                monto: monto,

                fecha: new Date().toLocaleString()

            });


            montoInput.value = '';


            // Si la deuda llega a 0 o menos, pasa al historial automáticamente

            if (calcularDeudaCliente(cliente) <= 0) {

                cliente.activo = false;

                alert('La deuda ha sido saldada completamente. El cliente pasa al Historial.');

                guardarStorage();

                switchTab('inicio');

                return;

            }


            guardarStorage();

            renderDetalle();

        }


        function pagarDeudaCompleta() {

            const cliente = db.clientes.find(c => c.id === clienteSeleccionadoId);

            const deudaActual = calcularDeudaCliente(cliente);


            if (deudaActual <= 0) return;


            if (confirm(`¿Confirmar pago total de $${deudaActual.toLocaleString()}?`)) {

                cliente.movimientos.push({

                    tipo: 'PAGO',

                    detalle: 'Pago Total Deuda',

                    monto: deudaActual,

                    fecha: new Date().toLocaleString()

                });

                cliente.activo = false;

                guardarStorage();

                switchTab('inicio');

            }

        }


        function mostrarResumenDeudores() {

            const deudores = db.clientes.filter(c => c.activo && calcularDeudaCliente(c) > 0);

            if (deudores.length === 0) {

                alert('No hay deudores activos en este momento.');

                return;

            }

            let resumen = "DEUDORES ACTIVOS:\n\n";

            deudores.forEach(d => {

                resumen += `• ${d.nombre}: $${calcularDeudaCliente(d).toLocaleString()}\n`;

            });

            alert(resumen);

        }


        // --- RENDERIZADO DE INTERFAZ ---

        function render() {

            // Render Productos

            const listProd = document.getElementById('lista-productos');

            listProd.innerHTML = db.productos.length === 0 ? '<p style="color:var(--subtext)">No hay productos</p>' :

                db.productos.map(p => `

                    <div class="list-item">

                        <span>${p.nombre}</span>

                        <strong>$${p.precio.toLocaleString()}</strong>

                    </div>

                `).join('');


            // Render Deudores Activos

            const listDeudores = document.getElementById('lista-deudores');

            let totalGlobal = 0;

            const activos = db.clientes.filter(c => c.activo);


            listDeudores.innerHTML = activos.length === 0 ? '<p style="color:var(--subtext)">No hay deudores activos</p>' :

                activos.map(c => {

                    const deuda = calcularDeudaCliente(c);

                    totalGlobal += deuda;

                    return `

                        <div class="list-item clickable" onclick="verDetalleCliente(${c.id})">

                            <div>

                                <strong>${c.nombre}</strong><br>

                                <small style="color:var(--subtext)">Toca para ver compras/pagos</small>

                            </div>

                            <strong style="color:var(--danger)">$${deuda.toLocaleString()}</strong>

                        </div>

                    `;

                }).join('');


            document.getElementById('global-total').innerText = `$${totalGlobal.toLocaleString()}`;


            // Render Historial

            const listHist = document.getElementById('lista-historial');

            const inactivos = db.clientes.filter(c => !c.activo);

            

            listHist.innerHTML = inactivos.length === 0 ? '<p style="color:var(--subtext)">Sin historial de pagos registrados</p>' :

                inactivos.map(c => `

                    <div class="list-item clickable" onclick="verDetalleCliente(${c.id})">

                        <div>

                            <strong>${c.nombre}</strong> <span class="badge" style="background:#E5V9E5; color:var(--success)">PAGADO</span><br>

                            <small style="color:var(--subtext)">Deuda completamente saldada</small>

                        </div>

                    </div>

                `).join('');

        }


        function renderDetalle() {

            if (!clienteSeleccionadoId) return;

            const cliente = db.clientes.find(c => c.id === clienteSeleccionadoId);

            const deuda = calcularDeudaCliente(cliente);

            

            document.getElementById('det-monto').innerText = `$${Math.max(0, deuda).toLocaleString()}`;


            const histDiv = document.getElementById('det-historial-movimientos');

            histDiv.innerHTML = cliente.movimientos.map(m => `

                <div class="list-item">

                    <div>

                        <strong>${m.detalle}</strong><br>

                        <small style="color:var(--subtext);">${m.fecha}</small>

                    </div>

                    <strong style="color: ${m.tipo === 'COMPRA' ? 'var(--danger)' : 'var(--success)'}">

                        ${m.tipo === 'COMPRA' ? '+' : '-'}$${m.monto.toLocaleString()}

                    </strong>

                </div>

            `).reverse().join('');

        }


        // Inicializar app

        render();
